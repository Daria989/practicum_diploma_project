SELECT 
    "Couriers".login, 
    COUNT("Orders".track) AS orders_count
FROM "Couriers"
LEFT OUTER JOIN "Orders" 
    ON "Couriers".id = "Orders"."courierId" 
    AND "Orders"."inDelivery" = true 
GROUP BY 
    "Couriers".login
ORDER BY 
    "Couriers".login;



SELECT 
    track,
    CASE 
        WHEN finished = true THEN 2
        WHEN cancelled = true THEN -1
        WHEN "inDelivery" = true THEN 1
        ELSE 0
    END AS status
FROM "Orders";
